## Decisions taken without asking (cheap to redo from the jsonl)

- **(a)'s reference when the record could not be read.** "Could not be read" means the record's open threw, or its
  read of any compared table threw; 118 of the 235 repaired plants are in that case. The reference is then the main
  file *as planted*, read alone. That file is `pristine.db`: 200 observations and 2 sessions, the same for every
  plant. The column "(a) vs input" compares instead with repair-db's own input: main+WAL after NEW doctor, or its
  main file alone when that read threw. The two columns give the same counts.
- **(d) coverage.** A missing rowid is accounted for when one of these holds:
  - it is in `r.tables[t].missing`;
  - it lies inside a `gaps` range of a table with `indexed === false`;
  - the table has `unreadable` set. This third case covered 0 rows.
- **(e) with an empty input log.** When the input log was absent or empty, the expected result is no kept `-wal`.
  repair-db does not keep an empty log.
- **"A refusal changed bytes"** means the sha1 of main or of `-wal` differs before and after. A file that went from
  present to absent counts as changed.

`plant()` are unchanged: 79 WAL pages x 11 variants = 869 plants (archived in
`docs/audits/20261021-v6.30.3-pretag-defects.md` Appendix A).

Changes against the first ruler:

- NEW `repair-db` is called **in-process** from the worker: `repairDb(<plant>/claude-mem-lite.db, { lockPath:
  <plant>/install.lock })` from `<NEW tree>/lib/db-repair.mjs` (`common.mjs` `repairInProc`). Its result is read
  directly: `outcome`, `exitCode`, `lines`, `tables[t] = { read, written, failed, gaps, missing, indexed, unreadable }`,
  `aside`. No stdout parsing. The default `swapGuard` is used (install.mjs passes its hook barrier; not exercised here).
- The worker process itself runs with the sandbox env: `sweep.sh` sets `D541_SLOT=w<k>`, and `run.sh` then gives it
  HOME/TMPDIR/CLAUDE_PROJECT_DIR under `sbx/w<k>-wrk` and CLAUDE_MEM_DIR = `rep/w<k>`, the directory each plant's repair
  copy is placed in.
- `sha()` is the full sha1. Loss coverage (`coveredBy`) comes from `r.tables`: a missing rowid is accounted for when it
  is in `missing`, inside a `gaps` range of a table with `indexed === false`, or in a table with `unreadable` set.
- (a) reference: rows of record (main+WAL as planted, before any doctor); when that read threw (open or any compared
  table), the main file as planted alone (`pristine.db`, read once in `probe.mjs` as `facts.pristineAlone`).
  `aIn`: against repair-db's input (main+WAL after NEW doctor, or its main file alone when that read threw).
- (e): the kept `<db>.damaged-*` equals the input main byte for byte, and its `-wal` equals the input `-wal` when that
  held bytes (absent when the input log was absent or empty).
- `probe-phantom.mjs <id>`: plants one plant and reads observations as the record does, to show whether the same
  bytes read the same way twice.

export function cks(buf, s0, s1, be) {
  for (let i = 0; i + 7 < buf.length; i += 8) {
    const a = be ? buf.readUInt32BE(i) : buf.readUInt32LE(i);
    const b = be ? buf.readUInt32BE(i + 4) : buf.readUInt32LE(i + 4);
    s0 = (s0 + a + s1) >>> 0;
    s1 = (s1 + b + s0) >>> 0;
  }
  return [s0, s1];
}
/** pristine main + the WAL with bytes [off, off+len) of the newest frame of `pageOf(db)` filled, chain re-signed, zero -shm. */
export function plant(dir, pageOf, off, len, fill) {
  const base = build();
  const p = fresh(dir);
  writeFileSync(p, readFileSync(join(base, 'pristine.db')));
  writeFileSync(p + '-wal', readFileSync(join(base, 'wal.orig')));
  const ro = new Database(p, { readonly: true });
  const pg = pageOf(ro);
  ro.close();
  writeFileSync(p, readFileSync(join(base, 'pristine.db')));
  rmSync(p + '-shm', { force: true });
  const wal = Buffer.from(readFileSync(join(base, 'wal.orig')));
  const be = wal.readUInt32BE(0) === 0x377f0683;
  const ps = wal.readUInt32BE(8);
  const frames = [];
  for (let o = 32; o + 24 + ps <= wal.length; o += 24 + ps) frames.push({ o, pg: wal.readUInt32BE(o) });
  const t = frames.filter((f) => f.pg === pg).at(-1);
  wal.fill(fill, t.o + 24 + off, t.o + 24 + off + len);
  let [s0, s1] = cks(wal.subarray(0, 24), 0, 0, be);
  for (const f of frames) {
    [s0, s1] = cks(wal.subarray(f.o, f.o + 8), s0, s1, be);
    [s0, s1] = cks(wal.subarray(f.o + 24, f.o + 24 + ps), s0, s1, be);
    wal.writeUInt32BE(s0, f.o + 16);
    wal.writeUInt32BE(s1, f.o + 20);
  }
  writeFileSync(p + '-wal', wal);
  writeFileSync(p + '-shm', Buffer.alloc(32768));
  return pg;
}
export function walPages() {
  const base = build();
  const wal = readFileSync(join(base, 'wal.orig'));
  const ps = wal.readUInt32BE(8);
  const pages = new Set();
  let frames = 0;
  for (let o = 32; o + 24 + ps <= wal.length; o += 24 + ps) {
    pages.add(wal.readUInt32BE(o));
    frames++;
  }
  return { pages: [...pages].sort((a, b) => a - b), frames, ps, walBytes: wal.length };
}

export const SHADOW_RE = /_fts_(data|idx|docsize|config|content)$/;
/** Ordinary tables: type='table', not a virtual table, not an FTS shadow table. */
export function ordinaryTables(db) {
  return db
    .prepare(`SELECT name, sql FROM sqlite_master WHERE type='table' ORDER BY name`)
    .all()
    .filter((r) => !/^CREATE VIRTUAL TABLE/i.test(r.sql || '') && !SHADOW_RE.test(r.name))
    .map((r) => r.name);
}
export function ftsTables(db) {
  return db
    .prepare(`SELECT name FROM sqlite_master WHERE type='table' AND sql LIKE 'CREATE VIRTUAL TABLE%' AND lower(sql) LIKE '%using fts5%' ORDER BY name`)
    .all()
    .map((r) => r.name);
}
const h1 = (s) => createHash('sha1').update(s).digest('hex');
/**
 * count, sha1 over JSON rows (SELECT * order) ordered by rowid — the archive's digest — and, when asked,
 * [rowid, rowHash] per row. Rows read before a throw are kept in `partial` (the digest is then {err}).
 */
export function tableDigest(db, t, { rows = false } = {}) {
  const rh = [];
  try {
    const it = db.prepare(`SELECT rowid AS __rid__, * FROM "${t}" ORDER BY rowid`).raw(true).iterate();
    const h = createHash('sha1');
    let n = 0;
    for (const full of it) {
      const r = full.slice(1);
      const j = JSON.stringify(r, (_k, v) => (typeof v === 'bigint' ? v.toString() : Buffer.isBuffer(v) ? 'x' + v.toString('hex') : v));
      h.update(j + '\n');
      n++;
      if (rows) rh.push([full[0], h1(j).slice(0, 12)]);
    }
    return { count: n, hash: h.digest('hex').slice(0, 16), ...(rows ? { rows: rh } : {}) };
  } catch (e) {
    return { err: `${e.code || ''} ${e.message}`.trim().slice(0, 160), ...(rows ? { partial: rh } : {}) };
  }
}
/** Copy main (+ wal, + shm if asked) into a fresh dir under WORK/tmp/<tag>; returns the copy path. */

}
export const ic = (db) => {
  try {
    const r = db.pragma('integrity_check').map((x) => x.integrity_check);
    return { ok: r.length === 1 && r[0] === 'ok', text: r.join(' ; ').slice(0, 220) };
  } catch (e) {
    return { ok: false, text: 'ERR ' + `${e.code || ''} ${e.message}`.trim().slice(0, 160) };
  }
};
/** Rows of a store: copy main+wal, open read-write on the copy (as Appendix A's readCopy), digest every ordinary table. */
export function readRows(dir, tag, tables, { rows = false } = {}) {
  const t = copyStore(dir, tag, { wal: true });
  let db;
  try {
    db = new Database(t);
  } catch (e) {
    return { openErr: `${e.code || ''} ${e.message}`.trim().slice(0, 160) };
  }
  try {
    let list = tables;
    if (!list) {
      try {
        list = ordinaryTables(db);
      } catch (e) {
        return { openErr: 'schema: ' + `${e.code || ''} ${e.message}`.trim().slice(0, 160) };
      }
    }
    const out = {};
    for (const tb of list) out[tb] = tableDigest(db, tb, { rows });
    return { tables: out };
  } finally {
    try {
      db.close();
    } catch {}
  }
}
/**
 * integrity_check on a copy of main alone (read-only handle), and on a copy of main + remaining wal
 * (read-write handle on the copy, as the archive: the files doctor left are never opened).
 */
export function integrity(dir, tag) {
  const res = {};
  {
    const t = copyStore(dir, tag + '-mo', { wal: false });
    try {
      const d = new Database(t, { readonly: true });
      res.mainOnly = ic(d);
      d.close();
    } catch (e) {
      res.mainOnly = { ok: false, text: 'OPENERR ' + e.message.slice(0, 120) };
    }
  }
  {
    const hasWal = existsSync(join(dir, DB + '-wal'));
    res.walPresent = hasWal;
    res.walBytes = hasWal ? readFileSync(join(dir, DB + '-wal')).length : 0;
    const t = copyStore(dir, tag + '-mw', { wal: true });
    try {
      const d = new Database(t);
      res.mainWal = ic(d);
      d.close();
    } catch (e) {
      res.mainWal = { ok: false, text: 'OPENERR ' + e.message.slice(0, 120) };
    }
  }
  return res;
}

export const digestOnly = (r) =>
  r.openErr
    ? { openErr: r.openErr }
    : { tables: Object.fromEntries(Object.entries(r.tables).map(([t, x]) => [t, x.err ? { err: x.err, partialCount: x.partial?.length } : { count: x.count, hash: x.hash }])) };
/** rows class vs a reference: kept (every table equal) / kept* (some reference table threw, every readable one equal) / lost. */
export function rowsClass(ref, post, { skip = [] } = {}) {
  if (ref.openErr) return { cls: 'ref-unreadable', diff: [] };
  const diff = [];
  let e = false;
  for (const [t, x] of Object.entries(ref.tables)) {
    if (skip.includes(t)) continue;
    if (x.err) {
      e = true;
      continue;
    }
    const q = post.tables?.[t];
    if (!q || q.err || q.count !== x.count || q.hash !== x.hash) diff.push(t);
  }
  return { cls: diff.length ? 'lost' : e ? 'kept*' : 'kept', diff };
}
/** Row-level diff of table t: rowids missing from / changed in / extra in `post` vs `ref` (needs rows: true reads). */
export function rowDiff(refT, postT) {
  const refRows = refT?.rows ?? refT?.partial ?? null;
  const postRows = postT?.rows ?? postT?.partial ?? [];
  if (!refRows) return null;
  const m = new Map(postRows.map(([id, h]) => [String(id), h]));
  const r = new Map(refRows.map(([id, h]) => [String(id), h]));
  const missing = [];
  const changed = [];
  for (const [id, h] of r) {
    if (!m.has(id)) missing.push(Number(id));
    else if (m.get(id) !== h) changed.push(Number(id));
  }
  const extra = [...m.keys()].filter((id) => !r.has(id)).map(Number);
  return { missing, changed, extra, refPartial: !refT.rows };
}
export const multisetHit = (have, want) => {
  const m = new Map();
  for (const [, h] of have || []) m.set(h, (m.get(h) || 0) + 1);
  let n = 0;
  for (const [, h] of want || []) if (m.get(h) > 0) (m.set(h, m.get(h) - 1), n++);
  return n;
};
/**
 * How repair-db's own result accounts for a rowid missing from the repaired store, table t (r.tables[t]):
 * 'missing' (listed in r.tables[t].missing), 'gap' (inside a gaps range on a table with indexed === false),
 * 'unreadable' (r.tables[t].unreadable set: the whole table reported unreadable), or null (silent).
 */
export const coveredBy = (rt, id) => {
  if (!rt) return null;
  if (rt.missing?.includes(id)) return 'missing';
  if (rt.unreadable) return 'unreadable';
  if (rt.indexed === false && rt.gaps?.some((g) => (g.after === null || id > g.after) && (g.before === null || id < g.before))) return 'gap';
  return null;
};
/** The main file alone (no log), on a copy: integrity_check and the observation count it reads. */

  copyFileSync(file, t);

    const d = new Database(t, { readonly: true });
    try {
      let obs;
      try {
        obs = d.prepare('SELECT count(*) c FROM observations').get().c;
      } catch (e) {
        obs = 'ERR ' + `${e.code || ''} ${e.message}`.trim().slice(0, 80);
      }

      if (tables) out.rows = { tables: Object.fromEntries(tables.map((tb) => [tb, tableDigest(d, tb, { rows: true })])) };

// Per plant: plant → NEW doctor (as in Q2) → three fresh copies of the post-doctor files, one step each:
//   S1 checkIntegrity (repair-db's first step, read-only), S2 probeExclusive (its second), S3 control: a plain
//   read-write open + SELECT 1 + close (what any hook or session connection does).
// Prints main/WAL sha and the main file alone (integrity, observation count) before and after each step.
import { join } from 'node:path';
import { appendFileSync } from 'node:fs';
import { WORK, TREES, DB, VARIANTS, Database, plant, doctor, sha, fresh, mainAlone } from './common.mjs';
import { copyFileSync, existsSync, rmSync } from 'node:fs';

const { checkIntegrity, probeExclusive } = await import(join(TREES.new, 'lib/db-repair.mjs'));
const steps = {
  S1_checkIntegrity: (p) => ({ ok: checkIntegrity(p).ok }),
  S2_probeExclusive: (p) => {
    const r = probeExclusive(p);
    return { acquired: r.acquired, busy: r.busy, err: r.error ? `${r.error.code} ${r.error.message}`.slice(0, 80) : null };
  },
  S3_plainRwOpenClose: (p) => {
    try {
      const d = new Database(p);
      let res;
      try {
        res = d.prepare('SELECT 1 x').get().x;
      } catch (e) {
        res = `${e.code} ${e.message}`.slice(0, 80);
      }
      d.close();
      return { res };
    } catch (e) {
      return { openErr: `${e.code} ${e.message}`.slice(0, 80) };
    }
  },
};
const out = join(WORK, 'out/raw/probe-checkpoint.jsonl');
for (const id of process.argv.slice(2)) {
  const [, pg, v] = /^p(\d+)-v(\d+)$/.exec(id).map(Number);
  const [off, len, fill] = VARIANTS[v];
  const dir = join(WORK, 'pc', 'plant');
  plant(dir, () => pg, off, len, fill);
  doctor('new', dir, 'pc-new');
  for (const [name, fn] of Object.entries(steps)) {
    const d = join(WORK, 'pc', name);
    fresh(d);
    for (const ext of ['', '-wal', '-shm']) if (existsSync(join(dir, DB + ext))) copyFileSync(join(dir, DB + ext), join(d, DB + ext));
    const p = join(d, DB);
    const before = { main: sha(p), wal: sha(p + '-wal'), mainAlone: mainAlone(p, 'pc-mo') };
    const res = fn(p);
    const after = { main: sha(p), wal: sha(p + '-wal'), mainAlone: mainAlone(p, 'pc-mo') };
    const row = { id, step: name, res, before, after, mainChanged: before.main !== after.main, walBefore: before.wal, walAfter: after.wal };
    appendFileSync(out, JSON.stringify(row) + '\n');
    console.log(`${id} ${name} ${JSON.stringify(res)} main ${before.main}→${after.main} wal ${before.wal}→${after.wal} mainAlone ${before.mainAlone.ic.ok ? 'ok' : 'bad'}/${before.mainAlone.obs} → ${after.mainAlone.ic.ok ? 'ok' : 'bad'}/${String(after.mainAlone.obs).slice(0, 50)}`);
  }
}
rmSync(join(WORK, 'pc'), { recursive: true, force: true });
```

// node probe-phantom.mjs <id> <mode>  — plant one plant, read observations as the record does (copy of main+WAL,
// read-write handle on the copy, SELECT rowid,* ORDER BY rowid), and print what came back: row count, distinct
// rowids, rowids outside the unplanted store's 1..241, and the error if the scan threw. Run several times in fresh
// processes to see whether one plant reads the same way every time.
import { join } from 'node:path';
import { WORK, VARIANTS, plant, readRows } from './common.mjs';
const [id] = process.argv.slice(2);
const [, pg, v] = /^p(\d+)-v(\d+)$/.exec(id).map(Number);
const [off, len, fill] = VARIANTS[v];
const dir = join(WORK, 'plants', 'phantom', id);
plant(dir, () => pg, off, len, fill);
const r = readRows(dir, 'phantom-rec', ['observations'], { rows: true });
const t = r.tables?.observations ?? {};
const rows = t.rows ?? t.partial ?? [];
const ids = rows.map(([id]) => Number(id));
const outside = ids.filter((x) => x < 1 || x > 241);
const hist = {};
for (const x of outside) hist[x] = (hist[x] || 0) + 1;
console.log(JSON.stringify({ id, rows: rows.length, distinct: new Set(ids).size, err: t.err ?? null, outside: hist }));
```
