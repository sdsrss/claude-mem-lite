run. It replaces the D#541 sweeps (20261021-v6.31.0-d541-sweep.md, and the round-2 defects review's
Appendix A) as the regression set the read strategy had to pass before doctor and the write-back were
wired: 0 rows lost without a line naming them and 0 rows SQLite returns intact left out, over the whole

- **Populations** (2,769 plants):

    of 200 observations; a log of 40 inserts, a VACUUM and 1 insert; one damaged log frame per plant,

  - `main:21:800`, `two:22:300`, `log:23:800`: the round-2 defects review's rich store (1,195

    migration cursors, `zz_values` with BLOB/INT64/REAL/NUL text, `zz_noindex` with no index), one

- **Per plant, on copies**: truth = the unplanted store's rows; readable = every truth row a rowid
  lookup, a forward scan or a backward scan returns, through main + log and through the main file alone,
  on a plain and on a writable_schema handle. Then repairDb in-process, and:
  - NOT-READ: a truth row SQLite returns with the truth's values (intact) that the repaired store lacks;
  - a truth row SQLite returns only with damaged values that the repaired store lacks is a note, not a
    problem, and must be named by id or by its table;
  - SILENT: a truth row the repaired store lacks with no line naming it (by id, by its table, or the
    whole-store lines for the log and for the list of tables);
  - CHANGED / PHANTOM: a row with other values, or a rowid the truth lacks, in a table no line names;
  - a refusal: the main file and the log byte for byte as before (an empty log or a -shm may appear), no
    other new file, and a next step printed;
  - a repair: integrity_check passes, and the kept files equal the main file and the log as found.

### After the round-1 repairs (1e0649eb), with a stricter ruler

The round-1 claims review mutation-tested this ruler: it could not see a damaged copy kept where an
intact one reads (it checked rowid presence only), a damage line withheld when another line names the
table, or a next step whose command is missing. The ruler was then changed: a table counts as named
only by a line that says rows may be missing or wrong; a next step needs a command line or an
explicit instruction; REPLACED counts rows kept with values other than the truth while SQLite returns
the truth's row. On 1e0649eb: repaired 1,639, refused 3, healthy 1,127; plants with a problem 11, all
REPLACED (garbled in a column no index covers, in a table named as damaged; 25 on c58637c2 with the
same ruler). doctor arm: 286 of 286 failing plants ✗, 0 of 583 passing, 0 written.

The round-2 review (20261022-v6.31.0-d546-r2.md) then found P1s this ruler cannot see: its truth is
the unplanted store's rows, its plants damage one page (or two leaves of one table), its stores have
no deletes below damaged rows in most tables and no table with only partial indexes and an interior
page, and it never damages a sqlite_master page of a WAL store's main file. A NOT-READ of 0 here is
therefore not evidence for those shapes.

## Defects of the read found by this ruler and fixed before c58637c2

- A skip-ahead seek that read only the rowid looped one rowid at a time on a row whose payload cannot
  be read (p63-v10, a 4 GB heap exhausted); the seek now reads the scan's columns.
- Taking the highest rowid read as the restart point let a garbage cell's rowid skip every real row
  after it; the restart point is the last rowid in scan order.
- A garbage cell under a real row's rowid hid the real row from its lookup (the rowid was marked read);
  a rowid counts as read once its row is written.
- A definition the log garbled but SQLite still parses shifted every later column (p10-v5); such a table
  is read through the main file's definition.
- A damaged index produced more than 100 problems and the per-table check's limit was counted against
  the table; SQLite checks the table's own tree first.

## NOT CHECKED

- Stores larger than the fixtures (a few MB); the round-2 review's 389 MB store was not re-run here.
- Rollback-journal stores, other page sizes, macOS and Windows.
- Plants with damage in both the main file and the log at once.
- The ruler's ability to say NO was not mutation-verified by the author; the round-3 claims review was
  asked to do it.

```js
export function rng(seed) {
  let s = seed >>> 0 || 1;
  return () => {
    s ^= s << 13;
    s >>>= 0;
    s ^= s >>> 17;
    s ^= s << 5;
    s >>>= 0;
    return s / 4294967296;
  };
}

function cks(buf, s0, s1, be) {
  for (let i = 0; i + 7 < buf.length; i += 8) {
    const a = be ? buf.readUInt32BE(i) : buf.readUInt32LE(i);
    const b = be ? buf.readUInt32BE(i + 4) : buf.readUInt32LE(i + 4);
    s0 = (s0 + a + s1) >>> 0;
    s1 = (s1 + b + s0) >>> 0;
  }
  return [s0, s1];
}

/** pristine main + the WAL with the newest frame of page `pg` filled, chain re-signed, zero -shm. */
export function plant869(dir, { pg, v }) {
  const base = build869();
  const [off, len, fill] = VARIANTS[v];
  const p = fresh(dir);
  writeFileSync(p, readFileSync(join(base, 'pristine.db')));
  const wal = Buffer.from(readFileSync(join(base, 'wal.orig')));
  const be = wal.readUInt32BE(0) === 0x377f0683;
  const ps = wal.readUInt32BE(8);
  const frames = [];
  for (let o = 32; o + 24 + ps <= wal.length; o += 24 + ps) frames.push({ o, pg: wal.readUInt32BE(o) });
  const t = frames.filter((f) => f.pg === pg).at(-1);
  wal.fill(fill, t.o + 24 + off, t.o + 24 + off + len);
  let [s0, s1] = cks(wal.subarray(0, 24), 0, 0, be);
  for (const f of frames) {
    [s0, s1] = cks(wal.subarray(f.o, f.o + 8), s0, s1, be);
    [s0, s1] = cks(wal.subarray(f.o + 24, f.o + 24 + ps), s0, s1, be);
    wal.writeUInt32BE(s0, f.o + 16);
    wal.writeUInt32BE(s1, f.o + 20);
  }
  writeFileSync(p + '-wal', wal);
  writeFileSync(p + '-shm', Buffer.alloc(32768));
  return p;
}

/** The rich template for a variant (main | two | log), built once per work dir. */
export function richTemplate(variant, { obs = 1200 } = {}) {
  const log = variant === 'log';
  const dir = join(WORK, `rich-${log ? 'log' : 'main'}-${obs}`);
  const p = join(dir, DB);
  if (existsSync(join(dir, 'done'))) return p;
  fresh(dir);
  const db = new Database(p);
  db.pragma('page_size = 4096');
  db.pragma('journal_mode = WAL');
  db.pragma('foreign_keys = OFF');
  initSchema(db);
  db.pragma('foreign_keys = OFF');
  ensureSearchTelemetrySchema(db);
  db.exec(`CREATE TABLE IF NOT EXISTS migration_cursors (name TEXT PRIMARY KEY, below_id INTEGER NOT NULL)`);
  db.prepare(`INSERT INTO migration_cursors VALUES ('obs-terms', 777)`).run();
  db.exec(`CREATE TABLE zz_values (id INTEGER PRIMARY KEY AUTOINCREMENT, b BLOB, big INTEGER, r REAL, t TEXT)`);
  db.exec(`CREATE TABLE zz_noindex (a TEXT, b INTEGER)`);
  for (let i = 0; i < 20; i++)
    db.prepare(
      `INSERT INTO sdk_sessions (content_session_id, memory_session_id, project, started_at, started_at_epoch) VALUES (?, ?, 'p', datetime('now'), 1)`,
    ).run(`c-s${i}`, `s${i}`);
  db.exec('BEGIN');
  fill(db, { obs });
  db.exec('COMMIT');
  db.prepare('DELETE FROM observations WHERE id > ?').run(obs - 5);
  db.pragma('wal_checkpoint(TRUNCATE)');
  db.close();
  if (log) {
    const raw = new Database(p);
    raw.pragma('foreign_keys = OFF');
    raw.exec('BEGIN');
    fill(raw, { obs: Math.floor(obs / 4), start: obs + 10, tag: 'late' });
    raw.exec('COMMIT');
    raw.prepare(`UPDATE observations SET title = title || ' updated' WHERE id % 50 = 0`).run();
    raw.prepare(`DELETE FROM user_prompts WHERE id % 97 = 0`).run();
    const walBytes = readFileSync(`${p}-wal`);
    const mainNow = readFileSync(p);
    raw.close();
    writeFileSync(p, mainNow);
    writeFileSync(`${p}-wal`, walBytes);
    rmSync(`${p}-shm`, { force: true });
  }
  writeFileSync(join(dir, 'done'), '');
  return p;
}

export function pageMap(p) {
  const db = new Database(p, { readonly: true });
  try {
    return db.prepare(`SELECT pageno, name, pagetype FROM dbstat ORDER BY pageno`).all();
  } finally {
    db.close();
  }
}
export function plantMain(p, pageno, offset, bytes, ps = 4096) {
  const fd = openSync(p, 'r+');
  try {
    writeSync(fd, Buffer.from(bytes), 0, bytes.length, (pageno - 1) * ps + offset);
  } finally {
    closeSync(fd);
  }
}
export function walFrames(p) {
  const wal = readFileSync(`${p}-wal`);
  const ps = wal.readUInt32BE(8);
  const frames = [];
  for (let o = 32; o + 24 + ps <= wal.length; o += 24 + ps) frames.push({ o, pg: wal.readUInt32BE(o) });
  return { frames, ps };
}
export function plantWal(p, frameIndex, offset, bytes) {
  const wal = Buffer.from(readFileSync(`${p}-wal`));
  const be = wal.readUInt32BE(0) === 0x377f0683;
  const ps = wal.readUInt32BE(8);
  const frames = [];
  for (let o = 32; o + 24 + ps <= wal.length; o += 24 + ps) frames.push({ o, pg: wal.readUInt32BE(o) });
  const f = frames[frameIndex];
  Buffer.from(bytes).copy(wal, f.o + 24 + offset);
  let [s0, s1] = cks(wal.subarray(0, 24), 0, 0, be);
  for (const fr of frames) {
    [s0, s1] = cks(wal.subarray(fr.o, fr.o + 8), s0, s1, be);
    [s0, s1] = cks(wal.subarray(fr.o + 24, fr.o + 24 + ps), s0, s1, be);
    wal.writeUInt32BE(s0, fr.o + 16);
    wal.writeUInt32BE(s1, fr.o + 20);
  }
  writeFileSync(`${p}-wal`, wal);
  rmSync(`${p}-shm`, { force: true });
  return f.pg;
}
export function copyPair(from, toDir) {
  const to = fresh(toDir);
  copyFileSync(from, to);
  if (existsSync(`${from}-wal`)) copyFileSync(`${from}-wal`, `${to}-wal`);
  return to;
}

/**
 * The plants of one rich sweep, generated exactly as r2-defects sweep.mjs does (same PRNG order), lazily:
 * returns [{ id, make(dir) → { p, plants } }].
 */
export function richPlants(variant, count, seed, slot = 'x') {
  const R = rng(seed);
  const pick = (a) => a[Math.floor(R() * a.length)];
  const tpl = richTemplate(variant);
  const pmap = pageMap(copyPair(tpl, join(WORK, 'scratch', slot, `pm-${variant}`)));
  const offsets = [
    { off: 0, bytes: () => Array(8).fill(0x41), what: 'hdr' },
    { off: 3, bytes: () => [0, 1], what: 'ncell1' },
    { off: 3, bytes: () => [0x7f, 0xff], what: 'ncellbig' },
    { off: 8, bytes: () => [0, 0, 0xff, 0xff], what: 'right-child/ptr' },
    { off: -1, bytes: () => Array.from({ length: 16 }, () => Math.floor(R() * 256)), what: 'rand16' },
    { off: -1, bytes: () => Array(64).fill(0x41), what: 'rand64' },
  ];
  const frames = variant === 'log' ? walFrames(tpl).frames : null;
  const out = [];
  for (let i = 0; i < count; i++) {
    // Draw this plant's parameters now, in the archived order.
    const steps = [];
    const n = variant === 'two' ? 2 : 1;
    const sameTable =
      variant === 'two' ? pick(pmap.filter((x) => x.pagetype === 'leaf' && !/_fts/.test(x.name))).name : null;
    for (let k = 0; k < n; k++) {
      const o = pick(offsets);
      const bytes = o.bytes();
      if (variant === 'log') {
        const last = new Map();
        frames.forEach((f, idx) => last.set(f.pg, idx));
        const fi = pick([...last.values()]);
        const off = o.off < 0 ? 100 + Math.floor(R() * 3900) : o.off + (frames[fi].pg === 1 ? 100 : 0);
        steps.push({ wal: true, fi, off, bytes, what: o.what, pg: frames[fi].pg });
      } else {
        const cand = sameTable
          ? pmap.filter((x) => x.name === sameTable && x.pagetype === 'leaf')
          : pmap.filter((x) => x.pageno > 1);
        const pg = pick(cand);
        const off = o.off < 0 ? Math.floor(R() * 4000) : o.off;
        steps.push({ wal: false, pg: pg.pageno, off, bytes, what: o.what });
      }
    }
    const desc = steps.map((s) => ({ pg: s.pg, off: s.off, what: s.what, name: pmap.find((x) => x.pageno === s.pg)?.name }));
    out.push({
      id: `${variant}${seed}-${i}`,
      desc,
      make(dir) {
        const p = copyPair(tpl, dir);
        for (const s of steps) {
          if (s.wal) plantWal(p, s.fi, s.off, s.bytes);
          else plantMain(p, s.pg, s.off, s.bytes);
        }
        return p;
      },
    });
  }
  return { tpl, plants: out };
}

// ---------------------------------------------------------------- oracle
const SHADOW = (name, virtual) => virtual.some((v) => name.startsWith(`${v}_`));
/** Ordinary tables (not virtual, not FTS shadow, not sqlite_stat*) → Map(rowid string → row JSON). */
export function record(p) {
  const db = new Database(p, { readonly: true, fileMustExist: true });
  try {
    const all = db.prepare(`SELECT name, sql FROM sqlite_master WHERE type='table'`).all();
    const virtual = all.filter((r) => /^CREATE VIRTUAL/i.test(r.sql || '')).map((r) => r.name);
    const out = {};
    for (const { name: t } of all) {
      if (virtual.includes(t) || SHADOW(t, virtual) || /^sqlite_stat/.test(t)) continue;
      const m = new Map();
      for (const r of db.prepare(`SELECT rowid AS __rowid, * FROM "${t}" ORDER BY rowid`).safeIntegers(true).iterate()) {
        const { __rowid, ...rest } = r;
        m.set(String(__rowid), serRow(rest));
      }
      out[t] = m;
    }
    return out;
  } finally {
    db.close();
  }
}

const openRo = (p, writable) => {
  const db = new Database(p, { readonly: true, fileMustExist: true });
  if (writable) {
    db.unsafeMode(true);
    db.pragma('writable_schema = ON');
  }
  return db;
};
/**
 * Which truth rows SQLite can give back from `p` (read-only, on a copy): per table, every truth rowid a
 * `WHERE rowid = ?` lookup reads, plus every row a forward and a backward scan read before throwing, each
 * tried on a plain handle and on a writable_schema handle. Returns { t: Map(rowid → row JSON) }.
 */
export function readable(p, tables) {
  const out = {};
  for (const writable of [false, true]) {
    let db;
    try {
      db = openRo(p, writable);
    } catch {
      continue;
    }
    try {
      for (const [t, ids] of Object.entries(tables)) {
        const m = (out[t] ??= new Map());
        let one;
        try {
          one = db.prepare(`SELECT rowid AS __rowid, * FROM "${t}" WHERE rowid = ?`).safeIntegers(true);
        } catch {
          continue;
        }
        const add = (r) => {
          if (!r) return;
          const { __rowid, ...rest } = r;
          const k = String(__rowid);
          if (!m.has(k)) m.set(k, serRow(rest));
        };
        for (const id of ids) {
          try {
            add(one.get(BigInt(id)));
          } catch {
            /* unreadable */
          }
        }
        for (const dir of ['', ' DESC']) {
          try {
            for (const r of db
              .prepare(`SELECT rowid AS __rowid, * FROM "${t}" ORDER BY rowid${dir}`)
              .safeIntegers(true)
              .iterate())
              add(r);
          } catch {
            /* stopped */
          }
        }
      }
    } finally {
      db.close();
    }
  }
  return out;
}
```

```js
// node run.mjs <population> <k> <K> <outFile>
//   population: 869 | main:<seed>:<count> | two:<seed>:<count> | log:<seed>:<count>
// Per plant (on copies of the planted files):
//   truth      = the unplanted store's rows (main + log), per ordinary table
//   mainTruth  = the unplanted main file's rows alone (log stores only)
//   readable   = truth rows a rowid lookup or a scan reads from main+log, and from the main file alone,
//                each on a plain and a writable_schema handle (plants.mjs readable())
// then repairDb in-process, and on its result:
//   notRead    = truth rowids readable from either source that the repaired store lacks
//   silent     = truth rowids the repaired store lacks that no line names
//   phantom/changed (unnamed) = rows not in the truth / differing from it, in a table no line names
//   refusals: main and log bytes unchanged, no new files but an empty -wal/-shm, a next step printed
//   repairs:  integrity ok, kept files byte-equal to the store as found
import { appendFileSync, copyFileSync, existsSync, readdirSync, rmSync, statSync } from 'node:fs';
import { join } from 'node:path';
import {
  DB, WORK, build869, copyPair, fresh, plant869, plants869, readable, record, richPlants, richTemplate, sha,
} from './plants.mjs';
const TREE = process.env.D546_TREE || new URL('../../../', import.meta.url).pathname;
const { repairDb, checkIntegrity } = await import(`${TREE}/lib/db-repair.mjs`);

const [pop, kS, KS, out] = process.argv.slice(2);
const k = Number(kS);
const K = Number(KS);
const slot = `w${k}`;

let plants;
let truth;
let mainTruth = null;
if (pop === '869') {
  const base = build869();
  truth = record(copyPair(join(base, 'unplanted', DB), join(WORK, 'scratch', slot, 'truth')));
  const mt = fresh(join(WORK, 'scratch', slot, 'mtruth'));
  copyFileSync(join(base, 'pristine.db'), mt);
  mainTruth = record(mt);
  plants = plants869().map((x) => ({ id: x.id, desc: x, make: (dir) => plant869(dir, x) }));
} else {
  const [variant, seedS, countS] = pop.split(':');
  const r = richPlants(variant, Number(countS), Number(seedS), slot);
  plants = r.plants;
  truth = record(copyPair(r.tpl, join(WORK, 'scratch', slot, 'truth')));
  if (variant === 'log') {
    const mt = fresh(join(WORK, 'scratch', slot, 'mtruth'));
    copyFileSync(r.tpl, mt);
    mainTruth = record(mt);
  }
}
const FRESH = await (async () => {
  const Database = (await import('better-sqlite3')).default;
  const { initSchema } = await import(`${TREE}/schema.mjs`);
  const m = new Database(':memory:');
  initSchema(m);
  const names = new Set(m.prepare(`SELECT name FROM pragma_table_list WHERE schema = 'main' AND type = 'table'`).pluck().all());
  m.close();
  return names;
})();
const truthIds = Object.fromEntries(Object.entries(truth).map(([t, m]) => [t, [...m.keys()]]));

// A line that says rows of the table may be missing or wrong (not any line that mentions it).
const LOSS = /(some of its pages are damaged|could not be read|an index lists|could not be written|from the database file without|definition reads differently|definition differs|disagree with its indexes)/;
const lineNames = (text, t) =>
  text.split('\n').some((l) => l.includes(`] ${t}: `) && LOSS.test(l)) || new RegExp(`[:,] ${t} \\(\\d+ rows;`).test(text);

for (let i = 0; i < plants.length; i++) {
  if (i % K !== k) continue;
  const pl = plants[i];
  const dir = join(WORK, 'plants', slot, 'cur');
  const p = pl.make(dir);
  const res = { id: pl.id, desc: pl.desc, problems: [] };
  const before = { main: sha(p), wal: sha(`${p}-wal`), walSize: statSync(`${p}-wal`, { throwIfNoEntry: false })?.size ?? 0, files: readdirSync(dir).sort() };
  res.integ = (() => {
    const c = checkIntegrity(copyPair(p, join(WORK, 'scratch', slot, 'ic')));
    return c.ok ? 'ok' : c.notChecked ? `notChecked: ${c.notChecked}` : 'fail';
  })();
  const rA = readable(copyPair(p, join(WORK, 'scratch', slot, 'ra')), truthIds);
  let rB = {};
  if (existsSync(`${p}-wal`) && statSync(`${p}-wal`).size > 0) {
    const d = fresh(join(WORK, 'scratch', slot, 'rb'));
    copyFileSync(p, d);
    rB = readable(d, truthIds);
  }
  const t0 = Date.now();
  let r;
  try {
    r = await repairDb(p, { lockPath: join(dir, 'install.lock') });
  } catch (e) {
    r = { outcome: 'THREW', lines: [{ level: 'fail', text: String(e?.stack ?? e) }] };
  }
  res.ms = Date.now() - t0;
  res.outcome = r.outcome;
  const text = r.lines.map((l) => `[${l.level}] ${l.text}`).join('\n');
  res.text = text.length > 3000 ? text.slice(0, 3000) : text;
  const globalLog = /could not read the database together with its log/.test(text);
  // A damaged list of tables: a table this version does not create, known only from that list,
  // cannot be named by name.
  const masterLine = /Part of its list of tables could not be read/.test(text);
  if (r.outcome === 'repaired') {
    if (!checkIntegrity(p).ok) res.problems.push('after: integrity fails');
    let after;
    try {
      after = record(p);
    } catch (e) {
      res.problems.push(`after unreadable: ${e.message}`);
      after = {};
    }
    const kept = readdirSync(dir).find((n) => /\.damaged-/.test(n) && !/-(wal|shm)$/.test(n));
    if (!kept) res.problems.push('no kept file');
    else {
      if (sha(join(dir, kept)) !== before.main) res.problems.push('kept main != as found');
      const keptWal = `${join(dir, kept)}-wal`;
      if (before.walSize > 0 && (!existsSync(keptWal) || sha(keptWal) !== before.wal)) res.problems.push('kept wal != as found');
    }
    for (const [t, m] of Object.entries(truth)) {
      if (t === 'schema_version') continue;
      const a = after[t] ?? new Map();
      const named = lineNames(text, t) || (masterLine && !FRESH.has(t));
      const req = new Set();
      for (const src of [rA, rB]) for (const id of src[t]?.keys() ?? []) if (m.has(id)) req.add(id);
      // A row SQLite gives back with the values the truth holds (intact), or with damaged ones.
      const intact = (id) => [rA, rB].some((src) => src[t]?.get(id) === m.get(id));
      const notRead = [...req].filter((id) => !a.has(id));
      const lostIntact = notRead.filter(intact);
      const lostDamaged = notRead.filter((id) => !intact(id));
      if (lostIntact.length && t !== 'sqlite_sequence') res.problems.push(`NOT-READ ${t}: ${lostIntact.length} (e.g. ${lostIntact.slice(0, 4).join(',')})`);
      if (lostDamaged.length && t !== 'sqlite_sequence') {
        const byId = lostDamaged.every((id) => new RegExp(`${t}: \\d+ row\\(s\\) could not be written[^\\n]*\\b${id}\\b`).test(text));
        res.notes = [...(res.notes ?? []), `read-damaged ${t}: ${lostDamaged.length} ${byId ? 'named by id' : 'named by table'}`];
      }
      const logOnly = (id) => mainTruth && !mainTruth[t]?.has(id);
      const silent = [...m.keys()].filter((id) => !a.has(id) && !named && !(globalLog && logOnly(id)));
      if (silent.length && t !== 'sqlite_sequence') res.problems.push(`SILENT ${t}: ${silent.length} of ${m.size} (e.g. ${silent.slice(0, 4).join(',')})`);
      const phantom = [...a.keys()].filter((id) => !m.has(id));
      if (phantom.length && t !== 'sqlite_sequence' && !named && !globalLog) res.problems.push(`PHANTOM ${t}: ${phantom.length} (e.g. ${phantom.slice(0, 3).join(',')})`);
      const changed = [...a.keys()].filter((id) => m.has(id) && a.get(id) !== m.get(id));
      // A damaged copy kept where SQLite also returns the row intact (r1 claims H1).
      const replaced = changed.filter(intact);
      if (replaced.length && t !== 'sqlite_sequence') res.problems.push(`REPLACED ${t}: ${replaced.length} (e.g. ${replaced.slice(0, 3).join(',')})`);
      if (changed.length && t !== 'sqlite_sequence' && !named && !globalLog) res.problems.push(`CHANGED ${t}: ${changed.length} (e.g. ${changed.slice(0, 3).join(',')})`);
      res.counts ??= {};
      if (t === 'observations') res.counts.obs = { truth: m.size, after: a.size, required: req.size };
    }
    for (const t of Object.keys(after)) if (!truth[t] && t !== 'schema_version') res.problems.push(`EXTRA TABLE ${t}`);
  } else if (r.outcome === 'refused') {
    const now = { main: sha(p), wal: sha(`${p}-wal`), files: readdirSync(dir).sort() };
    if (now.main !== before.main) res.problems.push('REFUSED but main changed');
    const EMPTY = 'da39a3ee5e6b4b0d3255bfef95601890afd80709';
    if (now.wal !== before.wal && !(before.wal === null && (now.wal === EMPTY || now.wal === null)) && !(before.wal === EMPTY && now.wal === null))
      res.problems.push(`REFUSED but wal changed`);
    const strip = (f) => f.filter((n) => !/-(wal|shm)$/.test(n) && n !== 'install.lock' && !/\.(damaged|repaired)-/.test(n));
    if (strip(now.files).join() !== strip(before.files).join()) res.problems.push(`files ${before.files} -> ${now.files}`);
    const command = r.lines.some((l) => l.level === 'info' && /^(node |touch |rm |ln |cp |mv )/.test(l.text));
    const nextStep = command || /repair-db again|free some space|repair-db of the newer/.test(text);
    if (!nextStep) res.problems.push('REFUSED with no next step');
    res.why = r.lines.filter((l) => l.level === 'fail').map((l) => l.text.slice(0, 200));
    let any = 0;
    for (const [t, m] of Object.entries(rA)) for (const id of m.keys()) if (truth[t]?.has(id)) any++;
    res.readable = any;
  } else if (r.outcome === 'healthy') {
    if (res.integ !== 'ok') res.problems.push(`healthy but integrity was ${res.integ}`);
  } else res.problems.push(`outcome ${r.outcome}`);
  appendFileSync(out, JSON.stringify(res) + '\n');
  rmSync(dir, { recursive: true, force: true });
}
```

