### F1 (P1) — the "disagreeing" rows are scan ordinals, not rowids

export const sha = (p) => {
  try {
    const b = fs.readFileSync(p);
    return b.length === 0 ? 'empty' : createHash('sha1').update(b).digest('hex');
  } catch {
    return 'absent';
  }

const J = (r) => JSON.stringify(r, (_, v) => (typeof v === 'bigint' ? `${v}` : v));

/** Ordinary tables (no FTS virtual/shadow, no sqlite_*) -> Map(rowid -> JSON). */
export function recordDb(db) {
  const virtual = db
    .prepare(`SELECT name FROM sqlite_master WHERE type='table' AND sql LIKE 'CREATE VIRTUAL TABLE%'`)
    .pluck()
    .all();
  const shadow = (n) => virtual.some((v) => n.startsWith(`${v}_`));
  const out = {};
  for (const t of db.prepare(`SELECT name FROM sqlite_master WHERE type='table'`).pluck().all()) {
    if (virtual.includes(t) || shadow(t) || t.startsWith('sqlite_')) continue;
    const m = new Map();
    for (const r of db.prepare(`SELECT rowid AS __rowid, * FROM "${t}" ORDER BY rowid`).safeIntegers(true).iterate())
      m.set(String(r.__rowid), J(r));
    out[t] = m;
  }
  try {
    out.__seq = new Map(db.prepare('SELECT name, seq FROM sqlite_sequence').raw(true).all().map(([n, s]) => [n, s]));
  } catch {
    out.__seq = new Map();
  }
  return out;
}
export function record(p) {
  const db = new Database(p, { readonly: true, fileMustExist: true });
  try {
    return recordDb(db);
  } finally {
    db.close();
  }
}

export function walFrames(wal) {
  const ps = wal.readUInt32BE(8);
  const frames = [];
  for (let o = 32; o + 24 + ps <= wal.length; o += 24 + ps)
    frames.push({ o, pg: wal.readUInt32BE(o), commit: wal.readUInt32BE(o + 4) });
  return { ps, frames };
}
function walChecksum(buf, s0, s1, be) {
  for (let i = 0; i + 7 < buf.length; i += 8) {
    const a = be ? buf.readUInt32BE(i) : buf.readUInt32LE(i);
    const b = be ? buf.readUInt32BE(i + 4) : buf.readUInt32LE(i + 4);
    s0 = (s0 + a + s1) >>> 0;
    s1 = (s1 + b + s0) >>> 0;
  }
  return [s0, s1];
}
export function resign(wal) {
  const be = wal.readUInt32BE(0) === 0x377f0683;
  const { ps, frames } = walFrames(wal);
  let [s0, s1] = walChecksum(wal.subarray(0, 24), 0, 0, be);
  for (const f of frames) {
    [s0, s1] = walChecksum(wal.subarray(f.o, f.o + 8), s0, s1, be);
    [s0, s1] = walChecksum(wal.subarray(f.o + 24, f.o + 24 + ps), s0, s1, be);
    wal.writeUInt32BE(s0, f.o + 16);
    wal.writeUInt32BE(s1, f.o + 20);
  }
}
/** Newest frame of page pg: its byte offset of the page image, or -1. */
export function newestFrame(wal, pg) {
  const { frames } = walFrames(wal);
  const f = frames.filter((x) => x.pg === pg).at(-1);
  return f ? f.o + 24 : -1;
}
export function plantMain(p, pg, off, bytes, ps = 4096) {
  const fd = fs.openSync(p, 'r+');
  try {
    fs.writeSync(fd, Buffer.from(bytes), 0, bytes.length, (pg - 1) * ps + off);
  } finally {
    fs.closeSync(fd);
  }
}
export function plantWal(p, pg, off, bytes) {
  const wal = Buffer.from(fs.readFileSync(`${p}-wal`));
  const at = newestFrame(wal, pg);
  if (at < 0) return false;
  Buffer.from(bytes).copy(wal, at + off);
  resign(wal);
  fs.writeFileSync(`${p}-wal`, wal);
  return true;
}
/** The newest frame of pg replaced by its previous version (an older frame, else the main file's page). */
export function staleWal(p, pg) {
  const wal = Buffer.from(fs.readFileSync(`${p}-wal`));
  const { ps, frames } = walFrames(wal);
  const fr = frames.filter((x) => x.pg === pg);
  if (fr.length === 0) return false;
  let older;
  if (fr.length > 1) older = Buffer.from(wal.subarray(fr.at(-2).o + 24, fr.at(-2).o + 24 + ps));
  else {
    const main = fs.readFileSync(p);
    if (main.length < pg * ps) return false;
    older = Buffer.from(main.subarray((pg - 1) * ps, pg * ps));
  }
  older.copy(wal, fr.at(-1).o + 24);
  resign(wal);
  fs.writeFileSync(`${p}-wal`, wal);
  return true;
}

/** Per-row reading of a store through rowid lookups (each row on its own), for "SQLite returns it intact". */
export function lookupRows(p, tables, ids) {
  const out = {};
  let db;
  try {
    db = new Database(p, { readonly: true, fileMustExist: true });
  } catch {
    return out;
  }
  try {
    for (const t of tables) {
      out[t] = new Map();
      let st;
      try {
        st = db.prepare(`SELECT rowid AS __rowid, * FROM "${t}" WHERE rowid = ?`).safeIntegers(true);
      } catch {
        continue;
      }
      for (const id of ids[t] ?? []) {
        try {
          const r = st.get(BigInt(id));
          if (r) out[t].set(id, J(r));
        } catch {
          /* unreadable */
        }
      }
    }
  } finally {
    db.close();
    fs.rmSync(`${p}-shm`, { force: true });
  }
  return out;
}

/** Parse repair-db lines into what each table is named for. */
export function parseLines(lines) {
  const named = new Map(); // table -> Set(kind)
  const ids = new Map(); // table -> Set(id)
  const truncated = new Set();
  const add = (t, k) => {
    if (!named.has(t)) named.set(t, new Set());
    named.get(t).add(k);
  };
  const addIds = (t, list) => {
    if (!ids.has(t)) ids.set(t, new Set());
    const m = /^(.*?) and (\d+) more$/.exec(list);
    if (m) truncated.add(t);
    for (const x of (m ? m[1] : list).split(/,\s*/)) ids.get(t).add(x.trim());
  };
  let global = new Set();
  for (const { text } of lines) {
    let m;
    if (/^SQLite could not read the database together with its log/.test(text)) global.add('logError');
    if (/^Part of its list of tables could not be read/.test(text)) global.add('masterIncomplete');
    if ((m = /^SQLite could not read (?:this table|these tables) with the log.*?\): (.*)\.$/s.exec(text))) {
      for (const part of m[1].matchAll(/(?:^|, )([^\s,(]+) \(\d+ rows;/g)) add(part[1], 'viaMain');
    }
    if ((m = /^(\S+): could not be read/.exec(text))) add(m[1], 'unreadable');
    if ((m = /^(\S+): some of its pages are damaged/.exec(text))) add(m[1], 'damaged');
    if ((m = /^(\S+): \d+ row\(s\) SQLite could not read with the log were taken/.exec(text))) add(m[1], 'fromMain');
    if ((m = /^(\S+): its definition differs/.exec(text))) add(m[1], 'defDiffers');
    if ((m = /^(\S+): it could not be made/.exec(text))) add(m[1], 'skipped');
    if ((m = /^(\S+): the internal table of a search index/.exec(text))) add(m[1], 'skippedShadow');
    if ((m = /^(\S+): a virtual table this version/.exec(text))) add(m[1], 'skippedVirtual');
    if ((m = /^(\S+): column\(s\)/.exec(text))) add(m[1], 'droppedCols');
    if ((m = /^(\S+): .* in the database with its log; its rows were read with the definition/.exec(text)))
      add(m[1], 'transplanted');
    if ((m = /^(\S+): \d+ row\(s\) an index lists could not be read \(ids (.*)\)\.$/.exec(text))) {
      add(m[1], 'missingIds');
      addIds(m[1], m[2]);
    }
    if ((m = /^(\S+): \d+ row\(s\) could not be written to the fresh file .*: ids (.*)\.$/.exec(text))) {
      add(m[1], 'notWritten');
      addIds(m[1], m[2]);
    }
    if ((m = /^(\S+): \d+ row\(s\) disagree with its indexes.*\(ids (.*)\)\.$/.exec(text))) {
      add(m[1], 'disagree');
      addIds(m[1], m[2]);
    }
    if ((m = /^(\S+): \d+ row\(s\) disagreed with its indexes.*\(ids (.*)\)\.$/.exec(text))) {
      add(m[1], 'replaced');
      addIds(m[1], m[2]);
    }
  }
  return { named, ids, truncated, global };
}

const TABLE_LEVEL = ['unreadable', 'damaged', 'fromMain', 'viaMain', 'defDiffers', 'skipped'];
const MAY_BE_BACK = ['fromMain', 'viaMain'];

/**
 * Judge a repaired store against the truth. Returns counts and examples.
 */
export function judge({ p, r, truth, mainAlone, asFoundIntact }) {
  const res = {
    outcome: r.outcome,
    integrity: null,
    lostUnnamed: [],
    changedUnnamed: [],
    backUnnamed: [],
    garbageUnnamed: [],
    intactDropped: [],
    intactAltered: [],
    lost: 0,
    changed: 0,
    back: 0,
    seqLow: [],
  };
  if (r.outcome !== 'repaired' && r.outcome !== 'unsound') return res;
  res.integrity = integrityOk(p);
  const after = record(p);
  const { named, ids, global } = parseLines(r.lines);
  const tableNamed = (t) =>
    global.has('logError') || [...(named.get(t) ?? [])].some((k) => TABLE_LEVEL.includes(k));
  const idNamed = (t, id) => ids.get(t)?.has(id);
  const mayBeBack = (t) => global.has('logError') || [...(named.get(t) ?? [])].some((k) => MAY_BE_BACK.includes(k));
  for (const [t, rows] of Object.entries(truth)) {
    if (t === '__seq' || t === 'schema_version') continue;
    const got = after[t];
    if (!got) {
      if (rows.size > 0 && !tableNamed(t) && !global.has('masterIncomplete'))
        res.lostUnnamed.push(`${t}:<table gone, ${rows.size} rows>`);
      res.lost += rows.size;
      continue;
    }
    for (const [id, v] of rows) {
      const g = got.get(id);
      if (g === undefined) {
        res.lost++;
        if (!tableNamed(t) && !idNamed(t, id)) res.lostUnnamed.push(`${t}:${id}`);
      } else if (g !== v) {
        res.changed++;
        if (!tableNamed(t) && !idNamed(t, id)) res.changedUnnamed.push(`${t}:${id}`);
      }
      const intact = asFoundIntact?.[t]?.has(id);
      if (intact) {
        if (g === undefined) res.intactDropped.push(`${t}:${id}`);
        else if (g !== v) res.intactAltered.push(`${t}:${id}`);
      }
    }
    for (const [id, g] of got) {
      if (rows.has(id)) continue;
      const wasInMain = mainAlone[t]?.has(id);
      if (wasInMain) {
        res.back++;
        if (!mayBeBack(t)) res.backUnnamed.push(`${t}:${id}`);
      } else if (!tableNamed(t) && !idNamed(t, id)) res.garbageUnnamed.push(`${t}:${id}`);
    }
  }
  for (const [name, seq] of truth.__seq) {
    const s = after.__seq.get(name);
    if (s === undefined || BigInt(s) < BigInt(seq)) res.seqLow.push(`${name}:${seq}->${s}`);
  }
  return res;
}

/**
 * Which truth rows SQLite returns intact: from the store as found (rowid lookup), or else from
 * the main file alone when the store as found does not give the row back intact.
 */
export function intactRows(p, truth, hasWal) {
  const tables = Object.keys(truth).filter((t) => t !== '__seq');
  const want = Object.fromEntries(tables.map((t) => [t, [...truth[t].keys()]]));
  const withLog = lookupRows(p, tables, want);
  const out = {};
  for (const t of tables) {
    out[t] = new Set();
    for (const [id, v] of truth[t]) if (withLog[t]?.get(id) === v) out[t].add(id);
  }
  if (hasWal) {
    const alone = `${p}.probe-alone`;
    fs.copyFileSync(p, alone);
    const rest = Object.fromEntries(tables.map((t) => [t, want[t].filter((id) => !out[t].has(id))]));
    const fromMain = lookupRows(alone, tables, rest);
    for (const ext of ['', '-wal', '-shm']) fs.rmSync(alone + ext, { force: true });
    for (const t of tables) for (const id of rest[t]) if (fromMain[t]?.get(id) === truth[t].get(id)) out[t].add(id);
  }
  return out;
}

export async function runRepair(treeName, p, opts = {}) {
  const T = await tree(treeName);
  const before = { main: sha(p), wal: sha(`${p}-wal`) };
  const t0 = Date.now();
  let r;
  try {
    r = await T.repairDb(p, { lockPath: join(dirname(p), 'install.lock'), ...opts });
  } catch (e) {
    r = { outcome: 'threw', lines: [{ level: 'fail', text: String(e?.stack ?? e) }], exitCode: 1 };
  }
  const ms = Date.now() - t0;
  const after = { main: sha(p), wal: sha(`${p}-wal`) };
  let bytesOk = null;
  if (r.outcome === 'refused' || r.outcome === 'threw') {
    bytesOk =
      after.main === before.main &&
      (after.wal === before.wal || (before.wal === 'absent' && after.wal === 'empty'));
  }
  const nextStep = r.lines.some((l) => l.level === 'info') ||
    r.lines.some((l) => /run repair-db again|chmod|Free some space|Close every|run:|Run the/.test(l.text));
  return { r, ms, before, after, bytesOk, nextStep };
}
